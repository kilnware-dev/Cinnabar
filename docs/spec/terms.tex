\chapter{Terms and Definitions}
\begin{list}{}{%
    \setlength{\leftmargin}{0pt}%
    \setlength{\parsep}{0pt}%
    \setlength{\itemsep}{-0.5em}%
    \setlength{\topsep}{0pt}%
}
    \item \subsection{expression}
    \para A construct of operators and values that produce a value.
    \subpara \(5 + 3\) evaluates to \(8\)
    \subsection{statement}
    \subpara \texttt{foo()} Evaluates to its return type. 
    \item
    \subsection{callable}
    \para Anything with an overloaded call operator.
    \subpara \texttt{fn foo() -> i32;}
    \subsection{call operator}
    \para Just a simple paranthese
    \subpara \texttt{()}

    \subsection{brackets}
    \para When speaking of brackets, they are always spoken of in pairs. The only exception is angle brackets.
    \subpara[Angle Brackets] \texttt{< >}
    \subpara[Brackets] \texttt{[ ]}
    \subpara[Curcly Brackets or Braces] \texttt{\{ \}}
    \subpara[Round Brackets or Paranthese] \texttt{( )}
\end{list}
